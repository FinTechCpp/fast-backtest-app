146 QString InteractiveChartView::createTooltipText(const QString& category, double value)
147 {
148  // Déterminer le type de période à partir de la catégorie
149  QString periodLabel;
150  QString detailedInfo = category; // Par défaut, utiliser la catégorie
151 
152  if (m_fullDates.contains(category)) {

154 
155  // Analyser le format de la catégorie pour déterminer le type de période
156  if (category.contains("/") && category.length() <= 10) {
157  // Format jour: "27/03/2025"
158  periodLabel = "Jour";
159  detailedInfo = referenceDate.toString("dddd dd MMMM yyyy");
160  }
161  else if (category.startsWith("S") && category.contains("(")) {

168  QDate weekEnd = weekStart.addDays(6);
169  detailedInfo = QString("S%1 de %2 (%3 au %4)")
170  .arg(weekNumber)
171  .arg(year)
172  .arg(weekStart.toString("dd MMMM"))
173  .arg(weekEnd.toString("dd MMMM"));
174  }
175  else if (category.contains("/") && category.length() == 7) {
176  // Format mois: "03/2025"
