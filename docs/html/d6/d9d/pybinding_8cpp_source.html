328 py::dict PyBindingManager::convertParamsToPython(const QMap<QString, QVariant>& params)
329 {
330  py::dict result;
331 
332  for (auto it = params.begin(); it != params.end(); ++it) {
333  const QString& key = it.key();
334  const QVariant& value = it.value();
335 
336  // Conversion selon le type QVariant
337  if (value.type() == QVariant::Bool) {
338  result[py::str(key.toStdString())] = py::bool_(value.toBool());
339  }
340  else if (value.type() == QVariant::Int) {
341  result[py::str(key.toStdString())] = py::int_(value.toInt());
342  }
343  else if (value.type() == QVariant::Double) {
344  result[py::str(key.toStdString())] = py::float_(value.toDouble());
345  }
346  else if (value.type() == QVariant::String) {
347  result[py::str(key.toStdString())] = py::str(value.toString().toStdString());
348  }
349  else if (value.type() == QVariant::Time) {
350  QTime time = value.toTime();
351  result[py::str(key.toStdString())] = py::str(time.toString("hh:mm:ss").toStdString());
352  }
353  else if (value.type() == QVariant::Date) {
354  QDate date = value.toDate();
355  result[py::str(key.toStdString())] = py::str(date.toString("dd/MM/yyyy").toStdString());
356  }
357  else if (value.type() == QVariant::List) {
358  py::list py_list;
359  QVariantList list = value.toList();
360  for (const QVariant& item : list) {
361  py_list.append(py::int_(item.toInt())); // Supposant que ce sont des entiers
362  }
363  result[py::str(key.toStdString())] = py_list;
364  }
365  else {
366  qWarning() << "Type non supporté pour la clé" << key << ":" << value.typeName();
367  // Convertir en string comme fallback
368  result[py::str(key.toStdString())] = py::str(value.toString().toStdString());
369  }
370  }
371 
372  return result;
373 }
