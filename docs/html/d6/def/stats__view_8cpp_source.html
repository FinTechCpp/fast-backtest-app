677 QString StatsView::formatDuration(const QVariant& value)
678 {
679  QString strValue = value.toString();
680 
681  // Si c'est déjà une chaîne formatée pandas (ex: "8 days 23:30:20")
682  if (strValue.contains("days") && strValue.contains(":")) {
683  // Parser le format pandas: "8 days 23:30:20"
684  QStringList parts = strValue.split(" ");
685  if (parts.size() >= 3) {
686  int days = parts[0].toInt();
687  QStringList timeParts = parts[2].split(":");
688  if (timeParts.size() >= 3) {
689  int hours = timeParts[0].toInt();
690  int minutes = timeParts[1].toInt();
691  int seconds = timeParts[2].toInt();
692 
693  return formatDetailedDuration(days, hours, minutes, seconds);
694  }
695  }
696  }
697 
698  // Si c'est "0 days 00:00:00" ou équivalent
699  if (strValue.contains("0 days 00:00:00") || strValue == "0") {
700  return "0sec";
701  }
702 
703  // Si c'est NaN ou vide
704  if (strValue.isEmpty() || strValue == "nan" || strValue == "NaT") {
705  return "N/A";
706  }
707 
708  // Essayer de convertir en nombre (peut-être en jours décimaux)
709  bool ok;
710  double numValue = value.toDouble(&ok);
711  if (ok && !std::isnan(numValue) && !std::isinf(numValue)) {
712  if (numValue == 0) {
713  return "0sec";
714  }
715 
716  // Convertir les jours décimaux en composants
717  int totalSeconds = static_cast<int>(numValue * 24 * 3600);
718  int days = totalSeconds / (24 * 3600);
719  totalSeconds %= (24 * 3600);
720  int hours = totalSeconds / 3600;
721  totalSeconds %= 3600;
722  int minutes = totalSeconds / 60;
723  int seconds = totalSeconds % 60;
724 
725  return formatDetailedDuration(days, hours, minutes, seconds);
726  }
727 
728  return strValue;
729 }
