221 {

224  for (auto it = m_widgets.begin(); it != m_widgets.end(); ++it) {
225  QWidget* widget = it.value();
226  QString key = it.key();
227 
228  if (QComboBox* combo = qobject_cast<QComboBox*>(widget)) {
229  values[key] = combo->currentIndex();
230  }
231  else if (QDoubleSpinBox* spinBox = qobject_cast<QDoubleSpinBox*>(widget)) {
232  values[key] = spinBox->value();

235  values[key] = spinBox->value();
236  }

238  values[key] = checkBox->isChecked();
239  }

241  values[key] = timeEdit->time().toString("hh:mm:ss");
242  }
243  }
244 
245  // Ajouter les paramètres use_atr_for_sl et use_atr_for_tp basés sur les méthodes sélectionnées

248  values["use_atr_for_sl"] = (slMethod->currentIndex() == 1); // Index 1 = ATR
249  }
250 
251  if (m_widgets.contains("tp_method")) {
252  QComboBox* tpMethod = static_cast<QComboBox*>(m_widgets["tp_method"]);
253  values["use_atr_for_tp"] = (tpMethod->currentIndex() == 1); // Index 1 = ATR
254  }
255 
256  // Ajouter la liste des jours de trading
257  QVariantList tradingDays;
258  for (int i = 0; i < 7; ++i) {

289  else if (QDoubleSpinBox* spinBox = qobject_cast<QDoubleSpinBox*>(widget)) {

300  if (time.isValid()) {

304  }
305  }
306 
307  // Traiter les jours de trading
308  if (values.contains("trading_days")) {
309  QVariantList tradingDays = values["trading_days"].toList();
310 
311  // Réinitialiser tous les jours
312  for (int i = 0; i < 7; ++i) {
313  QString key = QString("trading_day_%1").arg(i);
314  if (m_widgets.contains(key)) {
315  static_cast<QCheckBox*>(m_widgets[key])->setChecked(false);
316  }
