321

324  // Parser manuellement "2022-02-14 14:30:10+00:00"

327  (d[2].digitValue() * 10) + d[3].digitValue();
328  int month = (d[5].digitValue() * 10) + d[6].digitValue();
329  int day = (d[8].digitValue() * 10) + d[9].digitValue();
330  int hour = (d[11].digitValue() * 10) + d[12].digitValue();
331  int minute = (d[14].digitValue() * 10) + d[15].digitValue();
332  int second = (d[17].digitValue() * 10) + d[18].digitValue();
333 
334  QDateTime timestamp(QDate(year, month, day), QTime(hour, minute, second), Qt::UTC);
335 
336  if (!timestamp.isValid()) {
337  return nullptr;
338  }
339 
340  // Trouver les virgules et parser les valeurs
341  int comma1 = line.indexOf(',', 25);
342  int comma2 = line.indexOf(',', comma1 + 1);
343  int comma3 = line.indexOf(',', comma2 + 1);
344  int comma4 = line.indexOf(',', comma3 + 1);
345 
346  if (comma1 == -1 || comma2 == -1 || comma3 == -1 || comma4 == -1) {
347  comma4 = line.length(); // Pas de volume
348  }
349 
350  double open = line.midRef(comma1 + 1, comma2 - comma1 - 1).toDouble();
351  double high = line.midRef(comma2 + 1, comma3 - comma2 - 1).toDouble();
352  double low = line.midRef(comma3 + 1, comma4 - comma3 - 1).toDouble();
